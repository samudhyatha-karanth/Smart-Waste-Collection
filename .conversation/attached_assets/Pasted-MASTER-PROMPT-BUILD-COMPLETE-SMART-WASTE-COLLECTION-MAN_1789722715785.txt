10. DATABASE SCHEMA
==================================================

Create PostgreSQL migrations for the following schema.

Enable UUID generation:

CREATE EXTENSION IF NOT EXISTS pgcrypto;

TABLE: zones

CREATE TABLE zones (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name VARCHAR(150) NOT NULL UNIQUE,
  code VARCHAR(50) NOT NULL UNIQUE,
  description TEXT,
  active BOOLEAN NOT NULL DEFAULT TRUE,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

TABLE: users

CREATE TABLE users (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  full_name VARCHAR(150) NOT NULL,
  email VARCHAR(255) NOT NULL UNIQUE,
  phone VARCHAR(30),
  password_hash TEXT NOT NULL,
  role VARCHAR(20) NOT NULL CHECK (role IN ('citizen','collector','admin')),
  status VARCHAR(20) NOT NULL DEFAULT 'active'
    CHECK (status IN ('active','inactive','suspended')),
  address TEXT,
  zone_id UUID REFERENCES zones(id) ON DELETE SET NULL,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

TABLE: collection_points

CREATE TABLE collection_points (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  zone_id UUID NOT NULL REFERENCES zones(id) ON DELETE CASCADE,
  name VARCHAR(150) NOT NULL,
  address TEXT NOT NULL,
  latitude DECIMAL(10,7),
  longitude DECIMAL(10,7),
  waste_types JSONB NOT NULL DEFAULT '[]'::jsonb,
  active BOOLEAN NOT NULL DEFAULT TRUE,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

TABLE: bins

CREATE TABLE bins (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  collection_point_id UUID NOT NULL REFERENCES collection_points(id) ON DELETE CASCADE,
  bin_code VARCHAR(100) NOT NULL UNIQUE,
  waste_type VARCHAR(30) NOT NULL,
  capacity_liters INTEGER NOT NULL DEFAULT 120 CHECK (capacity_liters > 0),
  fill_percentage INTEGER NOT NULL DEFAULT 0 CHECK (fill_percentage BETWEEN 0 AND 100),
  status VARCHAR(30) NOT NULL DEFAULT 'normal'
    CHECK (status IN ('normal','near_full','overflowing','damaged','inactive')),
  last_reported_at TIMESTAMPTZ,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

TABLE: collection_schedules

CREATE TABLE collection_schedules (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  zone_id UUID NOT NULL REFERENCES zones(id) ON DELETE CASCADE,
  collection_point_id UUID NOT NULL REFERENCES collection_points(id) ON DELETE CASCADE,
  waste_type VARCHAR(30) NOT NULL,
  frequency VARCHAR(30) NOT NULL,
  days_of_week JSONB NOT NULL DEFAULT '[]'::jsonb,
  start_time TIME NOT NULL,
  end_time TIME NOT NULL,
  active BOOLEAN NOT NULL DEFAULT TRUE,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

TABLE: collection_assignments

CREATE TABLE collection_assignments (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  schedule_id UUID NOT NULL REFERENCES collection_schedules(id) ON DELETE CASCADE,
  collector_id UUID NOT NULL REFERENCES users(id) ON DELETE RESTRICT,
  collection_date DATE NOT NULL,
  status VARCHAR(30) NOT NULL DEFAULT 'assigned'
    CHECK (status IN (
      'scheduled',
      'assigned',
      'accepted',
      'in_progress',
      'completed',
      'missed',
      'unable_to_collect',
      'skipped',
      'cancelled'
    )),
  started_at TIMESTAMPTZ,
  completed_at TIMESTAMPTZ,
  notes TEXT,
  failure_reason TEXT,
  evidence_url TEXT,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

TABLE: complaints

CREATE TABLE complaints (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  citizen_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  assigned_collector_id UUID REFERENCES users(id) ON DELETE SET NULL,
  collection_assignment_id UUID REFERENCES collection_assignments(id) ON DELETE SET NULL,
  category VARCHAR(40) NOT NULL,
  title VARCHAR(120) NOT NULL,
  description TEXT NOT NULL,
  location TEXT NOT NULL,
  latitude DECIMAL(10,7),
  longitude DECIMAL(10,7),
  severity VARCHAR(20) NOT NULL DEFAULT 'medium'
    CHECK (severity IN ('low','medium','high','critical')),
  status VARCHAR(30) NOT NULL DEFAULT 'submitted'
    CHECK (status IN (
      'submitted',
      'acknowledged',
      'assigned',
      'in_progress',
      'resolved',
      'rejected'
    )),
  image_url TEXT,
  resolution_note TEXT,
  resolved_at TIMESTAMPTZ,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

TABLE: complaint_status_history

CREATE TABLE complaint_status_history (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  complaint_id UUID NOT NULL REFERENCES complaints(id) ON DELETE CASCADE,
  changed_by UUID NOT NULL REFERENCES users(id) ON DELETE RESTRICT,
  old_status VARCHAR(30),
  new_status VARCHAR(30) NOT NULL,
  note TEXT,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

TABLE: overflow_reports

CREATE TABLE overflow_reports (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  reported_by UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  bin_id UUID REFERENCES bins(id) ON DELETE SET NULL,
  location TEXT NOT NULL,
  latitude DECIMAL(10,7),
  longitude DECIMAL(10,7),
  severity VARCHAR(20) NOT NULL DEFAULT 'high'
    CHECK (severity IN ('low','medium','high','critical')),
  status VARCHAR(30) NOT NULL DEFAULT 'reported'
    CHECK (status IN ('reported','assigned','in_progress','resolved','rejected')),
  description TEXT,
  image_url TEXT,
  assigned_collector_id UUID REFERENCES users(id) ON DELETE SET NULL,
  resolved_at TIMESTAMPTZ,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

TABLE: notifications

CREATE TABLE notifications (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  type VARCHAR(50) NOT NULL,
  title VARCHAR(200) NOT NULL,
  message TEXT NOT NULL,
  related_entity_type VARCHAR(50),
  related_entity_id UUID,
  read_at TIMESTAMPTZ,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

TABLE: ai_advisories

CREATE TABLE ai_advisories (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES users(id) ON DELETE SET NULL,
  request_type VARCHAR(50) NOT NULL,
  input_text TEXT NOT NULL,
  result JSONB NOT NULL,
  model VARCHAR(100),
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

TABLE: audit_logs

CREATE TABLE audit_logs (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  actor_id UUID REFERENCES users(id) ON DELETE SET NULL,
  action VARCHAR(100) NOT NULL,
  entity_type VARCHAR(100) NOT NULL,
  entity_id UUID,
  metadata JSONB NOT NULL DEFAULT '{}'::jsonb,
  ip_address INET,
  user_agent TEXT,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

Create indexes on:

users(email)
users(role)
users(zone_id)

collection_assignments(collector_id)
collection_assignments(collection_date)
collection_assignments(status)

complaints(citizen_id)
complaints(status)
complaints(severity)
complaints(assigned_collector_id)

overflow_reports(status)
